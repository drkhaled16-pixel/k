import os from flask import Flask, jsonify, request, render_template_string from flask_sqlalchemy import SQLAlchemy app = Flask(__name__) app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///quran_course.db' app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False db = SQLAlchemy(app) # ========================================== # 1. نماذج قواعد البيانات (Database Models) # ========================================== class Student(db.Model): id = db.Column(db.Integer, primary_key=True) name = db.Column(db.String(100), nullable=False) academic_number = db.Column(db.String(20), unique=True, nullable=False) grades = db.relationship('Grade', backref='student', lazy=True, cascade="all, delete-orphan") class Unit(db.Model): id = db.Column(db.Integer, primary_key=True) title = db.Column(db.String(100), nullable=False) lessons = db.relationship('Lesson', backref='unit', lazy=True, cascade="all, delete-orphan") class Lesson(db.Model): id = db.Column(db.Integer, primary_key=True) title = db.Column(db.String(100), nullable=False) unit_id = db.Column(db.Integer, db.ForeignKey('unit.id'), nullable=False) quizzes = db.relationship('Quiz', backref='lesson', lazy=True, cascade="all, delete-orphan") class Quiz(db.Model): id = db.Column(db.Integer, primary_key=True) title = db.Column(db.String(100), nullable=False) max_score = db.Column(db.Float, default=10.0) lesson_id = db.Column(db.Integer, db.ForeignKey('lesson.id'), nullable=False) grades = db.relationship('Grade', backref='quiz', lazy=True, cascade="all, delete-orphan") class Grade(db.Model): id = db.Column(db.Integer, primary_key=True) student_id = db.Column(db.Integer, db.ForeignKey('student.id'), nullable=False) quiz_id = db.Column(db.Integer, db.ForeignKey('quiz.id'), nullable=False) score_achieved = db.Column(db.Float, nullable=False) # درجة الاختبار recitation_score = db.Column(db.Float, default=0.0) # درجة التلاوة والحفظ participation_score = db.Column(db.Float, default=0.0) # درجة المشاركة والتفاعل # ========================================== # 2. الواجهة الأمامية (HTML + CSS + JS) # ========================================== HTML_TEMPLATE = """ مادة القرآن الكريم وتفسيره - الأول الثانوي

مادة القرآن الكريم وتفسيره

منهج الصف الأول الثانوي - نظام إدارة المقرر والطلاب والدرجات

الوحدات والدروس المسجلة

إضافة وحدة أو درس جديد

إضافة طالب جديد

قائمة الطلاب

# اسم الطالب الرقم الأكاديمي التقرير

إنشاء اختبار لدرس

رصد درجات طالب

كشف الدرجات

الرقم الأكاديمي:

الاختبار درجة الاختبار درجة التلاوة والحفظ المشاركة المجموع الكلي
""" # ========================================== # 3. واجهات API (Backend Endpoints) # ========================================== @app.route('/') def index(): return render_template_string(HTML_TEMPLATE) @app.route('/api/units', methods=['GET']) def get_units(): units = Unit.query.all() result = [] for u in units: lessons = [] for l in u.lessons: quizzes = [{"id": q.id, "title": q.title, "max_score": q.max_score} for q in l.quizzes] lessons.append({"lesson_id": l.id, "lesson_title": l.title, "quizzes": quizzes}) result.append({"unit_id": u.id, "unit_title": u.title, "lessons": lessons}) return jsonify(result) @app.route('/api/lessons/add', methods=['POST']) def add_lesson(): data = request.json new_lesson = Lesson(title=data['title'], unit_id=int(data['unit_id'])) db.session.add(new_lesson) db.session.commit() return jsonify({"message": "تم إضافة الدرس بنجاح"}), 201 @app.route('/api/students', methods=['GET', 'POST']) def handle_students(): if request.method == 'POST': data = request.json new_student = Student(name=data['name'], academic_number=data['academic_number']) db.session.add(new_student) db.session.commit() return jsonify({"message": "تم إضافة الطالب بنجاح"}), 201 students = Student.query.all() return jsonify([{"id": s.id, "name": s.name, "academic_number": s.academic_number} for s in students]) @app.route('/api/quizzes', methods=['GET']) def get_quizzes(): quizzes = Quiz.query.all() return jsonify([{"id": q.id, "title": q.title, "max_score": q.max_score} for q in quizzes]) @app.route('/api/quizzes/add', methods=['POST']) def add_quiz(): data = request.json new_quiz = Quiz(title=data['title'], max_score=data.get('max_score', 10.0), lesson_id=int(data['lesson_id'])) db.session.add(new_quiz) db.session.commit() return jsonify({"message": "تم إحداث الاختبار بنجاح"}), 201 @app.route('/api/grades/assign', methods=['POST']) def assign_grade(): data = request.json grade = Grade( student_id=int(data['student_id']), quiz_id=int(data['quiz_id']), score_achieved=float(data['score_achieved']), recitation_score=float(data.get('recitation_score', 0.0)), participation_score=float(data.get('participation_score', 0.0)) ) db.session.add(grade) db.session.commit() return jsonify({"message": "تم حفظ الدرجة بنجاح"}), 201 @app.route('/api/students//report', methods=['GET']) def get_student_report(student_id): student = Student.query.get_or_404(student_id) grades = Grade.query.filter_by(student_id=student_id).all() report = { "student_name": student.name, "academic_number": student.academic_number, "grades_detail": [] } for g in grades: quiz = Quiz.query.get(g.quiz_id) report["grades_detail"].append({ "quiz_title": quiz.title if quiz else "اختبار محذوف", "quiz_score": g.score_achieved, "quiz_max": quiz.max_score if quiz else 0, "recitation_score": g.recitation_score, "participation_score": g.participation_score, "total_for_lesson": g.score_achieved + g.recitation_score + g.participation_score }) return jsonify(report) # ========================================== # 4. تهيئة البيانات والتأكيد # ========================================== def seed_database(): if not Unit.query.first(): # إضافة وحدات مقرر الأول الثانوي u1 = Unit(title="الوحدة الأولى: مدخل إلى التفسير وأصوله") u2 = Unit(title="الوحدة الثانية: علوم القرآن الكريم") u3 = Unit(title="الوحدة الثالثة: تفسير سورة الفاتحة والسور المقررة") db.session.add_all([u1, u2, u3]) db.session.commit() # إضافة دروس نموذجية l1 = Lesson(title="الدرس الأول: تعريف التفسير ونشأته", unit_id=u1.id) l2 = Lesson(title="الدرس الثاني: أقسام التفسير ومناهجه", unit_id=u1.id) l3 = Lesson(title="الدرس الثالث: الوحي ونزول القرآن الكريم", unit_id=u2.id) l4 = Lesson(title="الدرس الرابع: تفسير سورة الفاتحة", unit_id=u3.id) db.session.add_all([l1, l2, l3, l4]) db.session.commit() # إضافة اختبارات أولية q1 = Quiz(title="اختبار تقويمي: نشأة التفسير", max_score=10, lesson_id=l1.id) q2 = Quiz(title="اختبار قصير: سورة الفاتحة", max_score=10, lesson_id=l4.id) db.session.add_all([q1, q2]) db.session.commit() # إضافة طلاب كعينات تجريبية s1 = Student(name="عبدالله أحمد السالم", academic_number="104001") s2 = Student(name="محمد عمر الشمري", academic_number="104002") db.session.add_all([s1, s2]) db.session.commit() if __name__ == '__main__': with app.app_context(): db.create_all() seed_database() print("-> التطبيق يعمل الآن! افتح متصفحك على العنوان: http://127.0.0.1:5000") app.run(debug=True)
